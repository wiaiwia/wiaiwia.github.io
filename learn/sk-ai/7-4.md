    ```python

        return sorted(score_dict, key=lambda k: score_dict[k], reverse=True)[:3]
    ```

26. 리스트 삭제 시 주의 - 잘못된 예시와 올바른 예시

    ```python
    # ❌ 잘못된 예시: del i는 임시 변수만 삭제
    def delete_student_wrong(students, name):
        for i in students:
            if i.get("name") == name:
                del i
                return True
        return False

    # ✅ 올바른 예시 1: 슬라이스 할당으로 덮어쓰기
    def delete_student(students, name):
        original_count = len(students)
        students[:] = [s for s in students if s.get("name") != name]
        return len(students) < original_count

    # ✅ 올바른 예시 2: enumerate로 인덱스 삭제
    def delete_student_v2(students, name):
        for idx, item in enumerate(students):
            if item.get("name") == name:
                del students[idx]
                return True
        return False
    ```
